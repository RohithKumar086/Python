Tuples

🔒 Tuples — Immutable Lists

A tuple is like a list that cannot be modified after creation. Faster than lists and can be used as dictionary keys.

tuples.py
# Creating tuples
point = (3, 4)
rgb = (255, 128, 0)
single = (42,)          # ← comma required for single element!
empty = ()
from_list = tuple([1,2,3])

# Access (same as lists)
print(point[0])         # 3
print(rgb[-1])          # 0
print(rgb[1:])          # (128, 0)

# Unpacking — very powerful!
x, y = point             # x=3, y=4
r, g, b = rgb            # r=255, g=128, b=0

# Extended unpacking with *
first, *rest = (1,2,3,4,5)
print(first)  # 1
print(rest)   # [2, 3, 4, 5]

# Swap values using tuples
a, b = 10, 20
a, b = b, a              # a=20, b=10

# Tuple methods (only 2!)
t = (1,2,3,2,1)
print(t.count(2))       # 2
print(t.index(3))       # 2

# Tuples as dict keys (lists can't do this!)
locations = {
    (40.7, -74.0): "New York",
    (51.5, -0.1): "London"
}

# Named tuples — more readable
from collections import namedtuple
Point = namedtuple("Point", ["x", "y"])
p = Point(3, 4)
print(p.x, p.y)  # 3 4
💡

When to use tuples: For data that shouldn't change (coordinates, RGB colors, database records), function return values, and as dictionary keys.

Sets

🎯 Sets — Unique Collections

An unordered collection of unique elements with O(1) membership testing. Internally uses a hash table.

sets.py
# Creating sets
colors = {"red", "green", "blue"}
nums = set([1, 2, 2, 3, 3])  # {1, 2, 3} — duplicates removed!
empty = set()              # {} creates a dict, NOT a set!

# Adding & Removing
colors.add("yellow")        # add single element
colors.update(["black", "white"])  # add multiple
colors.remove("red")        # remove (KeyError if missing)
colors.discard("xyz")       # remove (NO error if missing)
item = colors.pop()          # remove & return arbitrary element

# ─── Set Operations (very useful for DSA!) ───
a = {1, 2, 3, 4}
b = {3, 4, 5, 6}

print(a | b)     # {1,2,3,4,5,6}  — Union (OR)
print(a & b)     # {3, 4}          — Intersection (AND)
print(a - b)     # {1, 2}          — Difference (in a, not in b)
print(a ^ b)     # {1, 2, 5, 6}    — Symmetric Difference (XOR)

# Subset / Superset
print({1,2} <= {1,2,3})  # True  — {1,2} is subset
print({1,2,3} >= {1,2})  # True  — {1,2,3} is superset

# ★ Fast membership test — O(1) vs O(n) for lists!
big_set = set(range(1000000))
print(999999 in big_set)  # True — instant!

# ★ DSA: Remove duplicates from list (preserving order)
data = [3,1,4,1,5,9,2,6,5]
seen = set()
unique = [x for x in data if not (x in seen or seen.add(x))]
print(unique)  # [3, 1, 4, 5, 9, 2, 6]

# ★ DSA: Find common elements
list1 = [1,2,3,4]
list2 = [3,4,5,6]
common = set(list1) & set(list2)
print(common)  # {3, 4}

# Set comprehension
sq_set = {x**2 for x in range(10)}
# {0, 1, 4, 9, 16, 25, 36, 49, 64, 81}

# frozenset — immutable set (can be dict key)
fs = frozenset([1,2,3])
