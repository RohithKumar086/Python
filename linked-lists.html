🔗 How Linked Lists Work

Unlike arrays, linked list elements are not stored contiguously in memory. Each element (node) contains data and a pointer to the next node.

🛠️ Singly Linked List — Full Implementation

linked_list.py
class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None

    # ─── Insert at the end ───
    def append(self, data):
        new_node = Node(data)
        if not self.head:
            self.head = new_node
            return
        curr = self.head
        while curr.next:
            curr = curr.next
        curr.next = new_node

    # ─── Insert at the beginning ───
    def prepend(self, data):
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node

    # ─── Delete by value ───
    def delete(self, data):
        if not self.head:
            return
        if self.head.data == data:
            self.head = self.head.next
            return
        curr = self.head
        while curr.next:
            if curr.next.data == data:
                curr.next = curr.next.next
                return
            curr = curr.next

    # ─── Search ───
    def search(self, data):
        curr = self.head
        while curr:
            if curr.data == data:
                return True
            curr = curr.next
        return False

    # ─── Reverse (in-place) ★ Very common interview Q! ───
    def reverse(self):
        prev = None
        curr = self.head
        while curr:
            next_node = curr.next
            curr.next = prev
            prev = curr
            curr = next_node
        self.head = prev

    # ─── Display ───
    def display(self):
        curr = self.head
        parts = []
        while curr:
            parts.append(str(curr.data))
            curr = curr.next
        print(" → ".join(parts) + " → None")

    # ─── Length ───
    def length(self):
        count = 0
        curr = self.head
        while curr:
            count += 1
            curr = curr.next
        return count

# ─── Usage ───
ll = LinkedList()
ll.append(10); ll.append(20); ll.append(30)
ll.prepend(5)
ll.display()     # 5 → 10 → 20 → 30 → None

ll.delete(20)
ll.display()     # 5 → 10 → 30 → None

ll.reverse()
ll.display()     # 30 → 10 → 5 → None

print(ll.search(10))  # True
print(ll.length())    # 3

🎯 Classic Linked List Problems

1. Detect Cycle (Floyd's Algorithm)

detect_cycle.py
def has_cycle(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next         # moves 1 step
        fast = fast.next.next    # moves 2 steps
        if slow == fast:
            return True          # cycle detected!
    return False

2. Find Middle Node

find_middle.py
def find_middle(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
    return slow  # slow is at the middle!

3. Merge Two Sorted Lists

merge_sorted.py
def merge_sorted(l1, l2):
    dummy = Node(0)
    curr = dummy
    while l1 and l2:
        if l1.data <= l2.data:
            curr.next = l1
            l1 = l1.next
        else:
            curr.next = l2
            l2 = l2.next
        curr = curr.next
    curr.next = l1 or l2
    return dummy.next
