💡 8 Essential Bit Hacks for Interviews

bit_tricks.py
# 1. Check if number is odd or even
def is_odd(n: int) -> bool:
    return (n & 1) == 1

# 2. Check if power of two
def is_power_of_two(n: int) -> bool:
    return n > 0 and (n & (n - 1)) == 0

# 3. Count set bits (Kernighan's Algorithm) — O(k) where k = number of 1s
def count_set_bits(n: int) -> int:
    count = 0
    while n > 0:
        n = n & (n - 1)  # Clears the lowest set bit!
        count += 1
    return count

# 4. Swap two numbers without extra variable
a, b = 5, 9
a = a ^ b
b = a ^ b
a = a ^ b
print(f"a={a}, b={b}")  # a=9, b=5

# 5. Get i-th bit of n
def get_bit(n: int, i: int) -> int:
    return (n >> i) & 1

# 6. Set i-th bit of n to 1
def set_bit(n: int, i: int) -> int:
    return n | (1 << i)

# 7. Clear i-th bit of n to 0
def clear_bit(n: int, i: int) -> int:
    return n & ~(1 << i)

# 8. Toggle i-th bit of n
def toggle_bit(n: int, i: int) -> int:
    return n ^ (1 << i)

1. Single Number (LeetCode #136)

Find the element that appears only once in an array where every other element appears twice. Solved in O(N) time and O(1) space using XOR!

single_number.py
def single_number(nums: list) -> int:
    res = 0
    for n in nums:
        res ^= n  # Paired numbers cancel out to 0!
    return res

print(single_number([4, 1, 2, 1, 2]))  # 4
