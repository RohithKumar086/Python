🕸️ What is a Graph?

A graph consists of vertices (nodes) and edges (connections). Graphs can be directed or undirected, weighted or unweighted.

Undirected Graph → Adjacency List
A → B, C
B → A, D
C → A, D
D → B, C

🛠️ Graph with Adjacency List

graph.py
from collections import defaultdict, deque

class Graph:
    def __init__(self):
        self.graph = defaultdict(list)

    def add_edge(self, u, v):
        self.graph[u].append(v)
        self.graph[v].append(u)  # remove for directed

    # ─── BFS — Breadth First Search ───
    # Uses a QUEUE → explores level by level
    def bfs(self, start):
        visited = set([start])
        queue = deque([start])
        result = []
        while queue:
            node = queue.popleft()
            result.append(node)
            for neighbor in self.graph[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
        return result

    # ─── DFS — Depth First Search (Iterative) ───
    # Uses a STACK → goes deep before backtracking
    def dfs_iterative(self, start):
        visited = set()
        stack = [start]
        result = []
        while stack:
            node = stack.pop()
            if node not in visited:
                visited.add(node)
                result.append(node)
                for neighbor in self.graph[node]:
                    if neighbor not in visited:
                        stack.append(neighbor)
        return result

    # ─── DFS — Recursive ───
    def dfs_recursive(self, node, visited=None):
        if visited is None:
            visited = set()
        visited.add(node)
        result = [node]
        for neighbor in self.graph[node]:
            if neighbor not in visited:
                result.extend(self.dfs_recursive(neighbor, visited))
        return result

    # ─── Has Path? ───
    def has_path(self, src, dest):
        visited = set()
        return self._dfs_path(src, dest, visited)

    def _dfs_path(self, node, dest, visited):
        if node == dest: return True
        visited.add(node)
        for neighbor in self.graph[node]:
            if neighbor not in visited:
                if self._dfs_path(neighbor, dest, visited):
                    return True
        return False

# ─── Usage ───
g = Graph()
g.add_edge("A", "B")
g.add_edge("A", "C")
g.add_edge("B", "D")
g.add_edge("C", "D")

print("BFS:", g.bfs("A"))           # ['A', 'B', 'C', 'D']
print("DFS:", g.dfs_iterative("A"))  # ['A', 'C', 'D', 'B']
print("Path A→D?", g.has_path("A","D"))  # True
