Fundamentals

📦 Variables & Assignment

A variable stores a value in memory. Python uses dynamic typing — no type declarations needed. The = operator assigns a value to a variable.

variables.py
# Assignment — the right side is evaluated first, then stored in the left
name = "Rohith"           # str  — text data
age = 21                   # int  — whole number
height = 5.9               # float — decimal number
is_student = True          # bool — True or False
nothing = None             # NoneType — no value

# Multiple assignment
x, y, z = 1, 2, 3
a = b = c = 0              # all set to 0

# Swap values (Python magic!)
x, y = y, x              # x=2, y=1

# Check type
print(type(name))         # <class 'str'>
print(type(age))          # <class 'int'>
print(isinstance(age, int)) # True

Variable Naming Rules

  • Must start with a letter or underscore (_)
  • Can contain letters, digits, and underscores
  • Case-sensitive — Age ≠ age
  • Cannot use Python keywords (if, for, class, etc.)
  • Use snake_case by convention — total_price, first_name
💡

Best Practice: Use descriptive names. count is better than c. student_name is better than sn.

Type Conversion (Casting)

type_conversion.py
# String → Number
x = int("42")        # 42
y = float("3.14")    # 3.14

# Number → String
s = str(100)         # "100"

# Number → Bool (0 is False, everything else is True)
print(bool(0))       # False
print(bool(1))       # True
print(bool(""))      # False  (empty string)
print(bool("hi"))    # True   (non-empty)
print(bool([]))      # False  (empty list)

# Conversions between collections
print(list("hello"))  # ['h', 'e', 'l', 'l', 'o']
print(tuple([1,2,3])) # (1, 2, 3)
print(set([1,2,2,3])) # {1, 2, 3}

operators.py
# Arithmetic deep dive
print(10 / 3)       # 3.3333...  (true division → float)
print(10 // 3)      # 3          (floor division → int)
print(10 % 3)       # 1          (remainder / modulo)
print(2 ** 10)      # 1024       (exponentiation)
print(-7 // 2)     # -4         (floors toward negative infinity)

# Comparison chaining — Python allows this!
x = 5
print(1 < x < 10)    # True  (same as 1 < x and x < 10)
print(1 < x < 3)     # False

# Logical operators — short-circuit evaluation
print(True and "hello")  # "hello"  (returns last truthy value)
print(False or "default") # "default" (returns first truthy value)
I/O

💬 Input & Output

io.py
# ─── print() ───
print("Hello", "World")              # Hello World
print("Hello", "World", sep="-")    # Hello-World
print("Hello", end=" ")               # No newline
print("World")                        # Hello World (same line)

# ─── input() — always returns a string! ───
name = input("Enter your name: ")
age = int(input("Enter your age: "))    # cast to int!

# ─── f-strings (Python 3.6+) — the best way to format! ───
print(f"Hi {name}, you are {age} years old!")
print(f"Next year: {age + 1}")
print(f"Pi = {3.14159:.2f}")            # Pi = 3.14 (2 decimals)
print(f"{'Hello':>20}")                # Right-align in 20 chars
print(f"{'Hello':*^20}")               # *******Hello********

# ─── .format() method ───
print("My name is {} and I am {}".format(name, age))
print("Price: ${:.2f}".format(49.9))     # Price: $49.90
Control Flow

🔀 If / Elif / Else

Python uses indentation (not braces) to define code blocks. Standard indent is 4 spaces.

conditionals.py
score = 85

if score >= 90:
    print("Grade: A — Excellent!")
elif score >= 80:
    print("Grade: B — Great job!")    # ← This runs
elif score >= 70:
    print("Grade: C — Good")
elif score >= 60:
    print("Grade: D — Pass")
else:
    print("Grade: F — Fail")

# Conditional (ternary) expression
status = "adult" if age >= 18 else "minor"

# Nested conditions
if score >= 80:
    if score == 100:
        print("Perfect score!")
    else:
        print("Great work!")

# Truthy / Falsy — these are all "Falsy":
# False, 0, 0.0, "", [], (), {}, set(), None
items = []
if not items:
    print("List is empty!")
Control Flow

🔁 Loops

While Loop

Repeats while a condition is True. Use when iteration count is unknown.

while.py
# Basic while loop
count = 1
while count <= 5:
    print(count, end=" ")
    count += 1
# Output: 1 2 3 4 5

# While with user input
while True:
    cmd = input("(q to quit): ")
    if cmd == "q":
        break
    print(f"You typed: {cmd}")

For Loop

Iterates over a sequence (list, string, range). Use when items are known.

for.py
# range(start, stop, step)
for i in range(1, 6):
    print(i, end=" ")
# Output: 1 2 3 4 5

# Iterate over a list
fruits = ["apple", "banana"]
for fruit in fruits:
    print(fruit)

# enumerate — index + value
for i, fruit in enumerate(fruits):
    print(f"{i}: {fruit}")

Nested Loops, Break & Continue

loop_controls.py
# Nested loop — multiplication table
for i in range(1, 4):
    for j in range(1, 4):
        print(f"{i}×{j}={i*j}", end="  ")
    print()
# 1×1=1  1×2=2  1×3=3
# 2×1=2  2×2=4  2×3=6
# 3×1=3  3×2=6  3×3=9

# break — exit the entire loop
for n in range(10):
    if n == 5: break
    print(n, end=" ")  # 0 1 2 3 4

# continue — skip the current iteration
for n in range(6):
    if n % 2 == 0: continue
    print(n, end=" ")  # 1 3 5

# for...else — else runs ONLY if loop completed without break
for n in [2, 4, 6]:
    if n % 2 != 0: break
else:
    print("All even!")  # This runs
Functions

⚙️ Functions

A function is a reusable block of code. Define once with def, call anywhere.

functions.py
# ─── Basic function ───
def greet(name):
    """Return a greeting string."""
    return f"Hello, {name}!"

print(greet("Rohith"))  # Hello, Rohith!

# ─── Default parameters ───
def power(base, exp=2):
    return base ** exp

print(power(3))       # 9  (uses default exp=2)
print(power(3, 3))    # 27

# ─── Multiple return values ───
def min_max(numbers):
    return min(numbers), max(numbers)

lo, hi = min_max([3, 1, 7, 2])
print(lo, hi)  # 1 7

# ─── Lambda (anonymous function) ───
square = lambda x: x ** 2
print(square(5))  # 25

# ─── *args — variable positional arguments ───
def total(*args):
    return sum(args)

print(total(1, 2, 3, 4))  # 10

# ─── **kwargs — variable keyword arguments ───
def info(**kwargs):
    for key, val in kwargs.items():
        print(f"{key}: {val}")

info(name="Rohith", age=21)
# name: Rohith
# age: 21
⚠️

Scope: Variables inside a function are local — they don't exist outside. Use global keyword (sparingly!) to modify a global variable from inside a function.
