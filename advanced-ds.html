Structure 1

🌲 Trie (Prefix Tree)

A tree-like structure used to efficiently store and search strings. Searching for a prefix of length L takes O(L) time regardless of how many words are in the dictionary!

trie.py
class TrieNode:
    def __init__(self):
        self.children = {}  # char → TrieNode
        self.is_end_of_word = False

class Trie:
    def __init__(self):
        self.root = TrieNode()

    # Insert word into Trie — O(L)
    def insert(self, word: str) -> None:
        curr = self.root
        for char in word:
            if char not in curr.children:
                curr.children[char] = TrieNode()
            curr = curr.children[char]
        curr.is_end_of_word = True

    # Search exact word — O(L)
    def search(self, word: str) -> bool:
        curr = self.root
        for char in word:
            if char not in curr.children:
                return False
            curr = curr.children[char]
        return curr.is_end_of_word

    # Check if any word starts with prefix — O(L)
    def starts_with(self, prefix: str) -> bool:
        curr = self.root
        for char in prefix:
            if char not in curr.children:
                return False
            curr = curr.children[char]
        return True

# Usage
trie = Trie()
trie.insert("apple")
print(trie.search("apple"))     # True
print(trie.search("app"))       # False
print(trie.starts_with("app"))  # True
Structure 2

🕸️ Disjoint Set Union (DSU / Union-Find)

Tracks elements partitioned into disjoint (non-overlapping) sets. With Path Compression and Union by Rank, operations run in nearly O(1) amortized time (inverse Ackermann function α(N))!

dsu.py
class UnionFind:
    def __init__(self, n: int):
        self.parent = list(range(n))  # Each element is its own parent initially
        self.rank = [1] * n            # Rank for union by rank

    # Find with Path Compression — O(α(N)) ≈ O(1)
    def find(self, i: int) -> int:
        if self.parent[i] != i:
            self.parent[i] = self.find(self.parent[i])  # Path compression!
        return self.parent[i]

    # Union by Rank — O(α(N)) ≈ O(1)
    def union(self, i: int, j: int) -> bool:
        root_i = self.find(i)
        root_j = self.find(j)

        if root_i != root_j:
            # Attach smaller rank tree under higher rank tree
            if self.rank[root_i] > self.rank[root_j]:
                self.parent[root_j] = root_i
            elif self.rank[root_i] < self.rank[root_j]:
                self.parent[root_i] = root_j
            else:
                self.parent[root_j] = root_i
                self.rank[root_i] += 1
            return True  # Union successful
        return False     # Already in same set (Cycle detected!)

# Usage — Detect cycle in undirected graph
uf = UnionFind(4)
print(uf.union(0, 1))  # True
print(uf.union(1, 2))  # True
print(uf.union(2, 0))  # False → Cycle detected between 0 and 2!
