How It Works

🗂️ What is a Dictionary?

A dictionary stores key-value pairs. Internally it uses a hash table — the key is hashed to find a bucket, giving O(1) average-case access.

How a Hash Map Works
key: "apple"→ hash()bucket[3]→value: 2
key: "banana"→ hash()bucket[7]→value: 5
key: "cherry"→ hash()bucket[1]→value: 8
📝

Keys must be immutable (str, int, tuple). Values can be anything. Keys must be unique — assigning to an existing key overwrites the value.

Creating & Accessing

🔨 Creating & Accessing Dictionaries

dict_basics.py
# Creating dictionaries
student = {
    "name": "Rohith",
    "age": 21,
    "courses": ["DSA", "Python"],
    "active": True
}
empty = {}
from_pairs = dict([("a", 1), ("b", 2)])  # {'a': 1, 'b': 2}
from_keys = dict.fromkeys(["x","y","z"], 0)  # {'x':0, 'y':0, 'z':0}

# ─── Accessing values ───
print(student["name"])                # Rohith
print(student.get("name"))            # Rohith
print(student.get("gpa", 0.0))       # 0.0  (default if key missing)
# student["gpa"] would raise KeyError!

# ─── Modifying ───
student["age"] = 22                  # update existing key
student["email"] = "r@mail.com"     # add new key
student.update({"gpa": 3.8, "age": 23})  # update multiple

# ─── Removing ───
del student["email"]                  # delete key (KeyError if missing)
val = student.pop("gpa")              # remove & return value
val = student.pop("xyz", None)        # safe pop with default
last = student.popitem()              # remove & return last pair

# setdefault — get value, or set it if missing
student.setdefault("country", "India")  # adds "country":"India"
Iteration

🔄 Iterating & Methods

dict_iteration.py
d = {"a": 1, "b": 2, "c": 3}

# Iterate over keys (default)
for key in d:
    print(key, d[key])

# Iterate over key-value pairs (most common)
for key, val in d.items():
    print(f"{key}: {val}")

# Just keys or values
print(list(d.keys()))    # ['a', 'b', 'c']
print(list(d.values()))  # [1, 2, 3]
print(list(d.items()))   # [('a',1), ('b',2), ('c',3)]

# Check if key exists
print("a" in d)         # True
print("x" in d)         # False

# Dictionary comprehension
squares = {x: x**2 for x in range(1, 6)}
# {1: 1, 2: 4, 3: 9, 4: 16, 5: 25}

# Invert a dictionary
inverted = {v: k for k, v in d.items()}
# {1: 'a', 2: 'b', 3: 'c'}
Collections Module

🧰 defaultdict, Counter & OrderedDict

collections_dict.py
from collections import defaultdict, Counter, OrderedDict

# ─── defaultdict — auto-creates missing keys ───
dd = defaultdict(list)
dd["fruits"].append("apple")     # no KeyError!
dd["fruits"].append("banana")
print(dd)  # {'fruits': ['apple', 'banana']}

word_count = defaultdict(int)
for word in "the cat sat on the mat".split():
    word_count[word] += 1
print(dict(word_count))
# {'the': 2, 'cat': 1, 'sat': 1, 'on': 1, 'mat': 1}

# ─── Counter — count occurrences ───
c = Counter("abracadabra")
print(c)                    # Counter({'a':5, 'b':2, 'r':2, 'c':1, 'd':1})
print(c.most_common(2))     # [('a', 5), ('b', 2)]
print(c["a"])               # 5
print(c["z"])               # 0  (no KeyError!)

# Counter arithmetic
c1 = Counter("aab")
c2 = Counter("abb")
print(c1 + c2)  # Counter({'a': 3, 'b': 3})
print(c1 - c2)  # Counter({'a': 1})

# ★ Two Sum (classic interview problem!)
def two_sum(nums, target):
    seen = {}  # value → index
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:
            return [seen[complement], i]
        seen[num] = i
    return []

print(two_sum([2,7,11,15], 9))  # [0, 1]  (2+7=9)

# ★ Group Anagrams
def group_anagrams(words):
    groups = defaultdict(list)
    for word in words:
        key = tuple(sorted(word))
        groups[key].append(word)
    return list(groups.values())

print(group_anagrams(["eat","tea","tan","ate","nat","bat"]))
# [['eat','tea','ate'], ['tan','nat'], ['bat']]

# ★ Frequency Sort
def freq_sort(s):
    freq = Counter(s)
    return "".join(c * n for c, n in freq.most_common())

print(freq_sort("tree"))  # "eert" or "eetr"
